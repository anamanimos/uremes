            function calculateAge(birthdateStr) {
                if (!birthdateStr) return 0;
                const birthDate = new Date(birthdateStr);
                const today = new Date();
                let age = today.getFullYear() - birthDate.getFullYear();
                const m = today.getMonth() - birthDate.getMonth();
                if (m < 0 || (m === 0 && today.getDate() < birthDate.getDate())) {
                    age--;
                }
                return age;
            }

            function validateCustomer() {
                const countryVal = document.getElementById('f_c8z').value;
                const requiredFields = [
                    { id: 'f_x7a', name: 'Nama Ketua' },
                    { id: 'f_h1b', name: 'Jenis Identitas' },
                    { id: 'f_v8n', name: 'Nomor Identitas' },
                    { id: 'f_t5g', name: 'Jenis Kelamin' },
                    { id: 'f_j4d', name: 'Tanggal Lahir' },
                    { id: 'f_m2q', name: 'No HP' },
                    { id: 'f_w6s', name: 'Alamat Lengkap' },
                    { id: 'f_c8z', name: 'Negara' },
                    { id: 'f_q7y', name: 'Metode Pembayaran' },
                    { id: 'tracking_hp', name: 'No HP Tracking' },
                    { id: 'tracking_pin', name: 'PIN Tracking' }
                ];

                if (countryVal === '99') {
                    requiredFields.push({ id: 'f_k3m', name: 'Provinsi' });
                    requiredFields.push({ id: 'f_r9p', name: 'Kota/Kabupaten' });
                }

                for (let field of requiredFields) {
                    const el = document.getElementById(field.id);
                    if (!el.value.trim()) {
                        Swal.fire({
                            icon: 'error',
                            title: 'Ada yang Terlewat',
                            text: `Mohon isi bagian "${field.name}" yang masih kosong pada Data Ketua.`,
                            confirmButtonColor: '#10b981'
                        });
                        el.focus();
                        el.classList.add('ring-2', 'ring-rose-500');
                        setTimeout(() => el.classList.remove('ring-2', 'ring-rose-500'), 2000);
                        return false;
                    }
                }

                const trackingPin = document.getElementById('tracking_pin').value;
                if (trackingPin.length < 4 || trackingPin.length > 6) {
                    Swal.fire({ icon: 'error', title: 'PIN Kurang Sesuai', text: 'Pastikan PIN Tracking Anda terdiri dari 4 sampai 6 angka.', confirmButtonColor: '#10b981' });
                    return false;
                }
                const ketuaAge = calculateAge(document.getElementById('f_j4d').value);
                if (ketuaAge < 10) {
                    Swal.fire({ icon: 'error', title: 'Belum Cukup Umur', text: 'Ketua rombongan harus berusia minimal 10 tahun.', confirmButtonColor: '#10b981' });
                    return false;
                }

                const memberCards = document.querySelectorAll('.member-card');
                if (memberCards.length < 1) {
                    Swal.fire({ icon: 'error', title: 'Pendaki Kurang', text: 'Anda tidak bisa mendaki sendirian. Minimal harus ada 2 orang (1 Ketua dan 1 Anggota). Silakan tambah anggota.', confirmButtonColor: '#10b981' });
                    return false;
                }

                const identityNumbers = new Set();
                const customerIdNo = document.getElementById('f_v8n').value;
                identityNumbers.add(customerIdNo);

                for (let i = 0; i < memberCards.length; i++) {
                    const card = memberCards[i];

                    const mName = card.querySelector('.mx_a').value;
                    const mBirthdate = card.querySelector('.mx_b').value;
                    const mIdentityType = card.querySelector('.mx_d').value;
                    const mIdentity = card.querySelector('.mx_e').value;
                    const mGender = card.querySelector('.mx_c').value;
                    const mHp = card.querySelector('.mx_f').value;
                    const mAddress = card.querySelector('.mx_j').value;
                    const mJob = card.querySelector('.mx_h').value;
                    const mFamilyHp = card.querySelector('.mx_g').value;
                    const mCountry = card.querySelector('.mx_i').value;

                    if (!mName || !mBirthdate || !mIdentityType || !mIdentity || !mGender || !mHp || !mAddress || !mJob || !mFamilyHp || !mCountry) {
                        Swal.fire({ icon: 'error', title: 'Data Anggota Belum Lengkap', text: `Ada bagian yang masih kosong pada Data Anggota ke-${i + 1}. Mohon lengkapi, atau klik hapus jika tidak diperlukan.`, confirmButtonColor: '#10b981' });
                        return false;
                    }

                    // Check for duplicate identity numbers
                    if (identityNumbers.has(mIdentity)) {
                        Swal.fire({
                            icon: 'error',
                            title: 'Identitas Ganda',
                            text: `Nomor identitas (KTP/Identitas lainnya) "${mIdentity}" pada Anggota ke-${i + 1} sudah terdaftar di rombongan ini. Harap periksa kembali.`,
                            confirmButtonColor: '#10b981'
                        });
                        return false;
                    }
                    identityNumbers.add(mIdentity);

                    if (mBirthdate) {
                        const mAge = calculateAge(mBirthdate);
                        if (mAge < 10) {
                            Swal.fire({ icon: 'error', title: 'Belum Cukup Umur', text: `Anggota ke-${i + 1} belum memenuhi usia minimum pendakian (10 tahun).`, confirmButtonColor: '#10b981' });
                            return false;
                        }
                        if (mAge > 70) {
                            const cb = card.querySelector('.mx_k');
                            if (!cb.checked) {
                                Swal.fire({ icon: 'error', title: 'Peringatan Lansia', text: `Anggota ke-${i + 1} berusia di atas 70 tahun. Tolong centang kotak persetujuan 'Surat Keterangan Dokter' pada data anggota tersebut.`, confirmButtonColor: '#10b981' });
                                return false;
                            }
                        }
                    }
                }

                return true;
            }

            async function finalizeAction(bookings) {
                const submitBtn = document.getElementById('bookingBtn');
                const originalBtnHtml = submitBtn.innerHTML;
                submitBtn.disabled = true;
                submitBtn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Mengirim data...';

                try {
                    const response = await fetch(API_BASE_URL + '/bookings', {
                        method: 'POST',
                        headers: {
                            'Content-Type': 'application/json',
                            'Accept': 'application/json'
                        },
                        body: JSON.stringify({ _p: obfuscate(bookings[0]) })
                    });

                    const result = await response.json();

                    if (response.ok && result.success) {
                        localStorage.removeItem('semeruBookingDraft');
                        Swal.fire({
                            icon: 'success',
                            title: 'Pendaftaran Sukses!',
                            html: `<p>Yeay! Formulir pendaftaran Anda berhasil terkirim.</p><p class="mt-2 font-mono bg-slate-100 px-3 py-2 rounded-lg text-sm">Kode Pendaftaran: <strong>${result.booking_id}</strong></p>`,
                            confirmButtonColor: '#10b981'
                        }).then(() => {
                            window.location.reload();
                        });
                    } else if (response.status === 422) {
                        const errors = result.errors ? Object.values(result.errors).flat().join('\n') : 'Ada data yang kurang tepat.';
                        Swal.fire({
                            icon: 'error',
                            title: 'Mohon Periksa Kembali',
                            text: errors,
                            confirmButtonColor: '#10b981'
                        });
                    } else {
                        throw new Error(result.message || 'Terjadi kesalahan sistem.');
                    }
                } catch (err) {
                    if (err.message === 'Failed to fetch') {
                        Swal.fire({
                            icon: 'error',
                            title: 'Koneksi Terputus',
                            text: 'Maaf, kami tidak dapat terhubung ke server. Pastikan koneksi internet Anda stabil atau coba beberapa saat lagi.',
                            confirmButtonColor: '#10b981'
                        });
                    } else {
                        Swal.fire({
                            icon: 'error',
                            title: 'Ups! Terjadi Kendala',
                            text: 'Maaf, saat ini sistem sedang memproses terlalu banyak permintaan. Silakan coba lagi nanti.',
                            confirmButtonColor: '#10b981'
                        });
                    }
                } finally {
                    submitBtn.disabled = false;
                    submitBtn.innerHTML = originalBtnHtml;
                }
            }

            document.getElementById('bookingBtn').addEventListener('click', () => {
                const terms = document.getElementById('termsCheckbox');
                if (!terms.checked) {
                    Swal.fire({ icon: 'warning', text: 'Tolong centang kotak persetujuan Persyaratan dan Kebijakan di bagian bawah.', confirmButtonColor: '#10b981' });
                    return;
                }

                if (!validateCustomer()) return;

                const targetDate = document.getElementById('f_d2x').value;
                const arrivalDate = document.getElementById('f_p6w').value;

                if (!targetDate || !arrivalDate) {
                    Swal.fire({ icon: 'error', text: 'Mohon pilih tanggal keberangkatan dan kepulangan Anda.', confirmButtonColor: '#10b981' });
                    return;
                }

                const totalMembers = document.querySelectorAll('.member-card').length;

                const ketuaCountry = document.getElementById('f_c8z').value;
                let wniCount = ketuaCountry === '99' ? 1 : 0;
                let wnaCount = ketuaCountry !== '99' && ketuaCountry !== '' ? 1 : 0;

                document.querySelectorAll('.member-card').forEach(card => {
                    const c = card.querySelector('.mx_i').value;
                    if (c === '99') wniCount++;
                    else if (c !== '') wnaCount++;
                });

                Swal.fire({
                    title: 'Konfirmasi Data',
                    html: `
                        <div class="text-left text-sm text-slate-600 space-y-3 mb-4 mt-2">
                            <div class="bg-slate-50 p-4 rounded-xl border border-slate-100">
                                <p class="mb-1"><strong>Ketua Kelompok:</strong> ${document.getElementById('f_x7a').value}</p>
                                <p class="mb-1"><strong>Keberangkatan:</strong> ${targetDate}</p>
                                <p class="mb-1"><strong>Kepulangan:</strong> ${arrivalDate}</p>
                                <p class="mb-1"><strong>Total Rombongan:</strong> ${totalMembers + 1} Orang</p>
                                <p class="text-xs mt-2 p-2 bg-white rounded border border-slate-200">
                                    <span class="inline-block mr-3"><i class="fas fa-flag text-rose-500"></i> WNI: <strong>${wniCount}</strong></span>
                                    <span class="inline-block"><i class="fas fa-globe-americas text-blue-500"></i> WNA: <strong>${wnaCount}</strong></span>
                                </p>
                            </div>
                            <div class="p-3 bg-brand-50 border border-brand-100 rounded-lg text-brand-700">
                                <i class="fas fa-info-circle mr-1"></i> Apakah Anda yakin semua data sudah benar?
                            </div>
                        </div>
                    `,
                    icon: 'question',
                    showCancelButton: true,
                    confirmButtonColor: '#10b981',
                    cancelButtonColor: '#94a3b8',
                    confirmButtonText: 'Ya, Data Benar',
                    cancelButtonText: 'Periksa Lagi'
                }).then((result) => {
                    if (result.isConfirmed) {
                        const bookings = [collectData("BOOKING_" + Date.now(), targetDate, arrivalDate)];
                        finalizeAction(bookings);
                    }
                });
            });

        });
